if(NOT TARGET nps_core)
    message(FATAL_ERROR "Configure nps with GIAC_HOST_ROOT to enable these tests")
endif()
set(GIAC_HOST_ARCHIVE "${GIAC_HOST_ROOT}/src/.libs/libgiac.a" CACHE FILEPATH "Built host Giac archive")
if(NOT EXISTS "${GIAC_HOST_ROOT}/src/config.h" OR NOT EXISTS "${GIAC_HOST_ARCHIVE}")
    message(FATAL_ERROR "Set GIAC_HOST_ROOT to a configured and built host Giac tree")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GMP REQUIRED IMPORTED_TARGET gmp)
find_library(MPFI_LIBRARY mpfi REQUIRED)
find_library(MPFR_LIBRARY mpfr REQUIRED)
find_path(MPFR_INCLUDE_DIR mpfr.h REQUIRED)
find_path(MPFI_INCLUDE_DIR mpfi.h REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(NPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(nps_giac_test_flags INTERFACE)
target_include_directories(nps_giac_test_flags INTERFACE ${NPS_ROOT}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/host_os)
target_include_directories(nps_giac_test_flags SYSTEM INTERFACE ${GIAC_HOST_ROOT}/src
    ${MPFR_INCLUDE_DIR} ${MPFI_INCLUDE_DIR})
target_compile_definitions(nps_giac_test_flags INTERFACE HAVE_CONFIG_H GIAC_GENERIC_CONSTANTS
    _LIBCPP_ENABLE_CXX17_REMOVED_BINDERS _LIBCPP_ENABLE_CXX17_REMOVED_UNARY_BINARY_FUNCTION)
target_compile_options(nps_giac_test_flags INTERFACE -fno-strict-aliasing -Wno-register)
target_link_libraries(nps_giac_test_flags INTERFACE ${GIAC_HOST_ARCHIVE} ${MPFI_LIBRARY}
    ${MPFR_LIBRARY} ${GMPXX_LIBRARY} PkgConfig::GMP Threads::Threads)
if(APPLE)
    find_library(ACCELERATE_FRAMEWORK Accelerate REQUIRED)
    target_link_libraries(nps_giac_test_flags INTERFACE ${ACCELERATE_FRAMEWORK})
endif()
add_library(nps_giac_host OBJECT ${NPS_ROOT}/src/cas/giac/giac_typed.cc)
target_link_libraries(nps_giac_host PRIVATE nps_giac_test_flags nps_flags)
foreach(probe matrix_capture guard_cleanup typed_differential matrix_cancellation determinant_capture)
    add_executable(${probe} ${probe}.cc)
    target_link_libraries(${probe} PRIVATE nps_core nps_manifest_host nps_giac_host nps_giac_test_flags)
    add_test(NAME ${probe} COMMAND ${probe})
endforeach()
add_executable(value_layout value_layout.cc)
target_link_libraries(value_layout PRIVATE nps_giac_test_flags nps_flags)
add_test(NAME value_layout COMMAND value_layout)
target_compile_definitions(guard_cleanup PRIVATE TIMEOUT)
target_compile_definitions(matrix_cancellation PRIVATE TIMEOUT)
target_compile_definitions(determinant_capture PRIVATE TIMEOUT)
pkg_check_modules(GIAC_TEST_LUA REQUIRED IMPORTED_TARGET luajit)
set_source_files_properties(${NPS_GIAC_SRC}/sha256.c PROPERTIES LANGUAGE CXX)
add_executable(lua_matrix lua_matrix.cc ${NPS_ROOT}/src/platform/nspire/integrity.cc
    ${NPS_GIAC_SRC}/sha256.c)
target_include_directories(lua_matrix BEFORE PRIVATE ${NPS_ROOT}/tests/target/host_os)
target_link_libraries(lua_matrix PRIVATE nps_core nps_manifest_host nps_giac_host
    nps_giac_test_flags PkgConfig::GIAC_TEST_LUA)
add_test(NAME lua_matrix COMMAND lua_matrix "${CMAKE_BINARY_DIR}/determinant-records.lua")
set_tests_properties(lua_matrix PROPERTIES FIXTURES_SETUP determinant_records)
